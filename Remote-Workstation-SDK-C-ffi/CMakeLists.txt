cmake_minimum_required(VERSION 3.24)
project(RemoteWorkstationSDKJsonSDK LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PRODUCT_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/native" CACHE PATH "Canonical SDK source")
if(NOT EXISTS "${PRODUCT_NATIVE_SOURCE}/CMakeLists.txt")
 set(PRODUCT_NATIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Remote-Workstation-SDK-CPP-vcpkg")
endif()
add_subdirectory("${PRODUCT_NATIVE_SOURCE}" native)
add_library(RemoteWorkstationSDKJsonSDK SHARED Bridge.cpp)
target_include_directories(RemoteWorkstationSDKJsonSDK PRIVATE "${PRODUCT_NATIVE_SOURCE}/sdk")
target_compile_features(RemoteWorkstationSDKJsonSDK PRIVATE cxx_std_20)
target_compile_options(RemoteWorkstationSDKJsonSDK PRIVATE /utf-8 /EHsc /W4)
target_compile_definitions(RemoteWorkstationSDKJsonSDK PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(RemoteWorkstationSDKJsonSDK PRIVATE RemoteWorkstationSDK::sdk)
