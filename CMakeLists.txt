cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(RemoteWorkstation VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build on Windows x64 with Visual Studio 2022 C++")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Separate user-mode memory instrumentation" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(RemoteTransport SDK/Transport.cpp)
target_link_libraries(RemoteTransport PUBLIC ws2_32 secur32 crypt32 ncrypt bcrypt)
add_library(RemoteSdk SDK/RemoteWorkstation.cpp SDK/License.cpp)
target_link_libraries(RemoteSdk PUBLIC RemoteTransport winhttp crypt32 advapi32 user32 gdi32 cabinet)
add_executable(RemoteWorkstation WIN32 App/Main.cpp)
target_link_libraries(RemoteWorkstation PRIVATE RemoteSdk comctl32 shell32)
add_executable(RemoteWorkstationCLI App/Console.cpp)
target_link_libraries(RemoteWorkstationCLI PRIVATE RemoteSdk)
add_executable(RemoteWorkstationExample Examples/view.cpp)
target_link_libraries(RemoteWorkstationExample PRIVATE RemoteSdk)
foreach(target RemoteWorkstation RemoteWorkstationCLI RemoteWorkstationExample)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(RemoteProtocolTests Tests/Protocol.cpp)
target_link_libraries(RemoteProtocolTests PRIVATE RemoteSdk)
add_executable(RemoteTransportTests Tests/Transport.cpp)
target_link_libraries(RemoteTransportTests PRIVATE RemoteTransport)
enable_testing()
add_test(NAME PinnedTlsLoopbackAndRefusal COMMAND RemoteTransportTests)
add_test(NAME FrameCodecAndInputBounds COMMAND RemoteProtocolTests)
set_tests_properties(PinnedTlsLoopbackAndRefusal PROPERTIES TIMEOUT 45)
set_target_properties(RemoteTransportTests RemoteProtocolTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(RemoteSetupCore InstallerUI/Setup.cpp InstallerUI/RuntimeSetup.cpp)
target_include_directories(RemoteSetupCore PRIVATE SDK)
target_link_libraries(RemoteSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32 shell32)
enable_language(RC)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed shared runtime package")
set(RUNTIME_RESOURCES "")
if(DRIVER_PACKAGE_DIRECTORY)
 foreach(name TitanSdkRuntime.inf TitanSdkRuntime.cat TitanSdkRuntime.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete opaque Titan SDK Runtime package")
  endif()
 endforeach()
 set(RUNTIME_RESOURCES "${CMAKE_CURRENT_BINARY_DIR}/Runtime.rc")
 file(WRITE "${RUNTIME_RESOURCES}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanSdkRuntime.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanSdkRuntime.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanSdkRuntime.sys\"\n")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/Application-$<CONFIG>.rc" CONTENT "201 RCDATA \"$<TARGET_FILE:RemoteWorkstation>\"\n202 RCDATA \"$<TARGET_FILE:RemoteWorkstationCLI>\"\n203 RCDATA \"$<TARGET_FILE:RemoteWorkstationExample>\"\n")
add_executable(RemoteWorkstationSetup WIN32 InstallerUI/Main.cpp "${CMAKE_CURRENT_BINARY_DIR}/Application-$<CONFIG>.rc" ${RUNTIME_RESOURCES})
add_executable(RemoteWorkstationSetupConsole InstallerConsole/Main.cpp "${CMAKE_CURRENT_BINARY_DIR}/Application-$<CONFIG>.rc" ${RUNTIME_RESOURCES})
foreach(target RemoteWorkstationSetup RemoteWorkstationSetupConsole)
 add_dependencies(${target} RemoteWorkstation RemoteWorkstationCLI RemoteWorkstationExample)
 target_link_libraries(${target} PRIVATE RemoteSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
